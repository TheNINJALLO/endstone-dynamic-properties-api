cmake_minimum_required(VERSION 3.20)
project(endstone_dynamic_properties VERSION 0.1.0 LANGUAGES CXX)

include(CMakePackageConfigHelpers)
include(GNUInstallDirs)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(ENDSTONE_DYNAMIC_PROPERTIES_BUILD_TESTS "Build portable tests" ON)
option(ENDSTONE_DYNAMIC_PROPERTIES_BUILD_EXAMPLES "Build portable examples" ON)
option(ENDSTONE_DYNAMIC_PROPERTIES_BUILD_PLUGIN "Build the Endstone plugin wrapper" OFF)
option(ENDSTONE_DYNAMIC_PROPERTIES_BUILD_NATIVE_2630 "Build the guarded BDS 26.30 boundary" ON)
option(ENDSTONE_DYNAMIC_PROPERTIES_VERIFIED_NATIVE_BRIDGE
       "Compile the generated, stage-verified complete-control bridge" OFF)
option(ENDSTONE_DYNAMIC_PROPERTIES_WARNINGS_AS_ERRORS
       "Treat project warnings as errors" ON)
set(ENDSTONE_BDS_BUILD "1.26.33" CACHE STRING "Exact Endstone runtime BDS build")
set(ENDSTONE_BDS_PACKAGE "1.26.33.1" CACHE STRING "Exact official BDS package version")

set(ENDSTONE_DYNAMIC_PROPERTIES_RELEASE_VERSION "${PROJECT_VERSION}")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/SOURCE_RELEASE.json")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/SOURCE_RELEASE.json"
         ENDSTONE_DYNAMIC_PROPERTIES_RELEASE_JSON)
    string(JSON ENDSTONE_DYNAMIC_PROPERTIES_RELEASE_VERSION
           GET "${ENDSTONE_DYNAMIC_PROPERTIES_RELEASE_JSON}" version)
endif()
string(REPLACE "." "\\."
       ENDSTONE_DYNAMIC_PROPERTIES_PROJECT_VERSION_REGEX "${PROJECT_VERSION}")
if(NOT ENDSTONE_DYNAMIC_PROPERTIES_RELEASE_VERSION MATCHES
   "^${ENDSTONE_DYNAMIC_PROPERTIES_PROJECT_VERSION_REGEX}($|[-+])")
    message(FATAL_ERROR
        "SOURCE_RELEASE.json version '${ENDSTONE_DYNAMIC_PROPERTIES_RELEASE_VERSION}' "
        "does not match project version '${PROJECT_VERSION}'")
endif()

add_library(endstone_dynamic_properties_core STATIC
    src/access_policy.cpp
    src/audit.cpp
    src/events.cpp
    src/in_memory_adapter.cpp
    src/native_binary_identity.cpp
    src/native_manifest.cpp
    src/operations.cpp
    src/service.cpp
    src/snapshot.cpp
    src/types.cpp
    src/value.cpp
)
add_library(EndstoneDynamicProperties::core ALIAS
    endstone_dynamic_properties_core)
set_target_properties(endstone_dynamic_properties_core PROPERTIES
    EXPORT_NAME core)
target_include_directories(endstone_dynamic_properties_core PUBLIC
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
    "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>"
)
target_compile_features(endstone_dynamic_properties_core PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(endstone_dynamic_properties_core PRIVATE /W4)
    if(ENDSTONE_DYNAMIC_PROPERTIES_WARNINGS_AS_ERRORS)
        target_compile_options(endstone_dynamic_properties_core PRIVATE /WX)
    endif()
else()
    target_compile_options(endstone_dynamic_properties_core PRIVATE -Wall -Wextra -Wpedantic)
    if(ENDSTONE_DYNAMIC_PROPERTIES_WARNINGS_AS_ERRORS)
        target_compile_options(endstone_dynamic_properties_core PRIVATE -Werror)
    endif()
endif()

if(ENDSTONE_DYNAMIC_PROPERTIES_VERIFIED_NATIVE_BRIDGE AND
   NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/verified_bds_26_30_adapter.cpp")
    message(FATAL_ERROR
        "Verified native bridge requested, but src/verified_bds_26_30_adapter.cpp "
        "was not generated by the activation workflow")
endif()

if(ENDSTONE_DYNAMIC_PROPERTIES_BUILD_PLUGIN)
    if(NOT ENDSTONE_BDS_BUILD STREQUAL "1.26.33" OR
       NOT ENDSTONE_BDS_PACKAGE STREQUAL "1.26.33.1")
        message(FATAL_ERROR
            "This package supports only BDS package 1.26.33.1/runtime 26.33 "
            "with Endstone v0.11.6")
    endif()
    if(NOT DEFINED ENDSTONE_CONAN_BDS_BUILD)
        message(FATAL_ERROR
            "ENDSTONE_CONAN_BDS_BUILD is missing; configure through this project's Conan toolchain")
    endif()
    if(NOT ENDSTONE_CONAN_BDS_BUILD STREQUAL ENDSTONE_BDS_BUILD)
        message(FATAL_ERROR
            "Conan dependencies target BDS ${ENDSTONE_CONAN_BDS_BUILD}, "
            "but CMake targets ${ENDSTONE_BDS_BUILD}")
    endif()

    find_package(base64 REQUIRED)
    find_package(Boost REQUIRED)
    find_package(concurrentqueue REQUIRED)
    find_package(cpptrace REQUIRED)
    find_package(date REQUIRED)
    find_package(EnTT REQUIRED)
    find_package(expected-lite REQUIRED)
    find_package(funchook REQUIRED)
    find_package(glm REQUIRED)
    find_package(magic_enum REQUIRED)
    find_package(Microsoft.GSL REQUIRED)
    find_package(nlohmann_json REQUIRED)
    find_package(raknet REQUIRED)
    find_package(replxx REQUIRED)
    find_package(sentry REQUIRED)
    find_package(spdlog REQUIRED)
    find_package(tomlplusplus REQUIRED)
    find_package(zstr REQUIRED)

    include(FetchContent)
    FetchContent_Declare(endstone
        GIT_REPOSITORY https://github.com/EndstoneMC/endstone.git
        GIT_TAG v0.11.6
    )
    FetchContent_MakeAvailable(endstone)

    set(DYNAMIC_PROPERTIES_PLUGIN_SOURCES src/plugin.cpp)
    if(ENDSTONE_DYNAMIC_PROPERTIES_BUILD_NATIVE_2630)
        list(APPEND DYNAMIC_PROPERTIES_PLUGIN_SOURCES src/bds_26_30_adapter.cpp)
    endif()
    if(ENDSTONE_DYNAMIC_PROPERTIES_VERIFIED_NATIVE_BRIDGE)
        list(APPEND DYNAMIC_PROPERTIES_PLUGIN_SOURCES
             src/verified_bds_26_30_adapter.cpp)
    endif()

    endstone_add_plugin(dynamic_properties_api ${DYNAMIC_PROPERTIES_PLUGIN_SOURCES})
    set_target_properties(dynamic_properties_api PROPERTIES
        PREFIX ""
        OUTPUT_NAME "endstone_dynamic_properties_bds_1_26_33"
        CXX_VISIBILITY_PRESET hidden
        VISIBILITY_INLINES_HIDDEN ON
    )
    target_include_directories(dynamic_properties_api PRIVATE include)
    target_link_libraries(dynamic_properties_api PRIVATE
        endstone_dynamic_properties_core
        endstone::endstone
    )
    target_compile_definitions(dynamic_properties_api PRIVATE
        ENDSTONE_DYNAMIC_PROPERTIES_NATIVE_2630=$<BOOL:${ENDSTONE_DYNAMIC_PROPERTIES_BUILD_NATIVE_2630}>
        ENDSTONE_DYNAMIC_PROPERTIES_VERIFIED_NATIVE_BRIDGE=$<BOOL:${ENDSTONE_DYNAMIC_PROPERTIES_VERIFIED_NATIVE_BRIDGE}>
        ENDSTONE_DYNAMIC_PROPERTIES_BDS_BUILD="${ENDSTONE_BDS_BUILD}"
        ENDSTONE_DYNAMIC_PROPERTIES_BDS_PACKAGE="${ENDSTONE_BDS_PACKAGE}"
        ENDSTONE_DYNAMIC_PROPERTIES_ENDSTONE_VERSION="0.11.6"
        ENDSTONE_DYNAMIC_PROPERTIES_VERSION="${ENDSTONE_DYNAMIC_PROPERTIES_RELEASE_VERSION}"
    )
endif()

if(ENDSTONE_DYNAMIC_PROPERTIES_BUILD_TESTS)
    enable_testing()
    add_executable(dynamic_properties_tests tests/cpp/test_dynamic_properties.cpp)
    target_link_libraries(dynamic_properties_tests PRIVATE
        endstone_dynamic_properties_core)
    if(MSVC)
        target_compile_options(dynamic_properties_tests PRIVATE /W4)
        if(ENDSTONE_DYNAMIC_PROPERTIES_WARNINGS_AS_ERRORS)
            target_compile_options(dynamic_properties_tests PRIVATE /WX)
        endif()
    else()
        target_compile_options(dynamic_properties_tests PRIVATE
            -Wall -Wextra -Wpedantic)
        if(ENDSTONE_DYNAMIC_PROPERTIES_WARNINGS_AS_ERRORS)
            target_compile_options(dynamic_properties_tests PRIVATE -Werror)
        endif()
    endif()
    add_test(NAME dynamic_properties_tests COMMAND dynamic_properties_tests)

    if(CMAKE_CONFIGURATION_TYPES)
        set(ENDSTONE_DYNAMIC_PROPERTIES_CONSUMER_MULTI_CONFIG TRUE)
    else()
        set(ENDSTONE_DYNAMIC_PROPERTIES_CONSUMER_MULTI_CONFIG FALSE)
    endif()
    add_test(
        NAME dynamic_properties_installed_consumer
        COMMAND "${CMAKE_COMMAND}"
            "-DPACKAGE_BUILD_DIR=${CMAKE_BINARY_DIR}"
            "-DCONSUMER_SOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}/tests/consumer"
            "-DWORK_ROOT=${CMAKE_CURRENT_BINARY_DIR}/_consumer/$<CONFIG>"
            "-DGENERATOR=${CMAKE_GENERATOR}"
            "-DGENERATOR_PLATFORM=${CMAKE_GENERATOR_PLATFORM}"
            "-DGENERATOR_TOOLSET=${CMAKE_GENERATOR_TOOLSET}"
            "-DMULTI_CONFIG=${ENDSTONE_DYNAMIC_PROPERTIES_CONSUMER_MULTI_CONFIG}"
            "-DCONFIG=$<CONFIG>"
            "-DCTEST_COMMAND=${CMAKE_CTEST_COMMAND}"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/consumer/RunInstalledConsumer.cmake"
    )
    set_tests_properties(dynamic_properties_installed_consumer PROPERTIES
        LABELS packaging)
endif()

if(ENDSTONE_DYNAMIC_PROPERTIES_BUILD_EXAMPLES)
    add_executable(dynamic_properties_complete_example
        examples/cpp/complete_control.cpp)
    target_link_libraries(dynamic_properties_complete_example PRIVATE
        endstone_dynamic_properties_core)
endif()

set(ENDSTONE_DYNAMIC_PROPERTIES_CMAKE_INSTALL_DIR
    "${CMAKE_INSTALL_LIBDIR}/cmake/EndstoneDynamicProperties")

configure_package_config_file(
    cmake/EndstoneDynamicPropertiesConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/EndstoneDynamicPropertiesConfig.cmake"
    INSTALL_DESTINATION "${ENDSTONE_DYNAMIC_PROPERTIES_CMAKE_INSTALL_DIR}"
)
write_basic_package_version_file(
    "${CMAKE_CURRENT_BINARY_DIR}/EndstoneDynamicPropertiesConfigVersion.cmake"
    VERSION "${PROJECT_VERSION}"
    COMPATIBILITY SameMajorVersion
)

install(TARGETS endstone_dynamic_properties_core
    EXPORT EndstoneDynamicPropertiesTargets
    ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}"
    LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}"
    RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}"
)
install(EXPORT EndstoneDynamicPropertiesTargets
    FILE EndstoneDynamicPropertiesTargets.cmake
    NAMESPACE EndstoneDynamicProperties::
    DESTINATION "${ENDSTONE_DYNAMIC_PROPERTIES_CMAKE_INSTALL_DIR}"
)
install(FILES
    "${CMAKE_CURRENT_BINARY_DIR}/EndstoneDynamicPropertiesConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/EndstoneDynamicPropertiesConfigVersion.cmake"
    DESTINATION "${ENDSTONE_DYNAMIC_PROPERTIES_CMAKE_INSTALL_DIR}"
)
install(DIRECTORY include/endstone_dynamic_properties
    DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
install(FILES
    README.md
    LICENSE
    NOTICE
    CHANGELOG.md
    BUILD_STATUS.md
    VALIDATION_RESULTS.md
    SOURCE_RELEASE.json
    DESTINATION .
)
install(DIRECTORY docs compatibility native tools examples DESTINATION .
    PATTERN "__pycache__" EXCLUDE
    PATTERN "*.pyc" EXCLUDE
)
install(DIRECTORY python/endstone_dynamic_properties DESTINATION python
    PATTERN "__pycache__" EXCLUDE
    PATTERN "*.pyc" EXCLUDE
)
